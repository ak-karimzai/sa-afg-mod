Lnew_cnt:
  ; in: 88@ handle; adds 1 to 60@ if that char exists and is alive
  00D6 IF 0
  056D DOES_CHAR_EXIST 88@
  004D GOTO_IF_FALSE Lnew_cnt_r
  00D6 IF 0
  NOT 0118 IS_CHAR_DEAD 88@
  004D GOTO_IF_FALSE Lnew_cnt_r
  000A ADD_VAL_TO_INT_LVAR 60@ 1
  0051 RETURN

  ; dead or gone: drop its blip (91@ in/out)

  NOT 0039 IS_INT_LVAR_EQUAL_TO_NUMBER 91@ -1
  004D GOTO_IF_FALSE Lnew_cnt_x
  0164 REMOVE_BLIP 91@
